# INC AWQ calibration followed by packed WQLinear INT4 CUDA inference.
# Activate Python >=3.11 with neural-compressor==3.10 and the existing awq==0.1.0
# checkout; rebuild awq/kernels for that interpreter/PyTorch/CUDA environment.
# --use_se_block loads the SE checkpoint; calibration remains independent of SE gates.
CUDA_VISIBLE_DEVICES=4 python test.py \
       --dataset ACDC \
       --vit_name R50-ViT-B_16 \
       --img_size 224 \
       --acdc_zspacing 5 \
       --repeated_runs 10 \
       --benchmark_dict benchmark_inc_awq_packed/ \
       --use_se_block \
       --quantize --quantize_backend inc_awq_packed \
       --quantize_calibrate_batch_size 8 \
       --ckpt_dir ckpt/ckpt_ACDC \
       --ckpt SELoss_ACDC__best_val_dice_0.8944082856178284_epoch_40_20260930_150207.pth \
       --description "INC activation-aware AWQ calibration with packed WQLinear INT4 CUDA inference"

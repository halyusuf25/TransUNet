# custom_w4 requires the official MIT-HAN-Lab AWQ CUDA runtime.
# Each dataset uses the same calibration-only SE checkpoint for both saliency modes.
# Replace REPLACE_WITH_ENDOVIS_SE_AUX_CHECKPOINT.pth before running EndoVis2018.

##############################
# Synapse: SE-aux saliency
##############################
CUDA_VISIBLE_DEVICES=0 python test.py \
  --dataset Synapse \
  --vit_name R50-ViT-B_16 \
  --img_size 224 \
  --batch_size 24 \
  --repeated_runs 10 \
  --benchmark_dict benchmark_quantize/ \
  --ckpt_dir ckpt/ckpt_Synapse/ \
  --ckpt SE_AUX_Calib_Synapse__best_val_dice_0.7875882387161255_epoch_97_20260709_092104.pth \
  --use_se_block \
  --se_calib_only \
  --quantize \
  --quant_backend custom_w4 \
  --saliency_source se_aux \
  --quantize_calibrate_batch_size 8 \
  --description "Synapse custom W4 quantization with SE-aux saliency"

Gumbel-TopK

         --dataset Cataract1k \
         --vit_name R50-ViT-B_16 \
         --max_epoch 270 \
         --batch_size 24 \
         --img_size 224 \

Gumbel-TopK + BU Loss

CUDA_VISIBLE_DEVICES=1 python train.py \
         --dataset Cataract1k \
         --vit_name R50-ViT-B_16 \
         --max_epoch 270 \
         --batch_size 24 \
         --img_size 224 \
         --topk_attn 0.50 \
         --use_gumbel_topk \
         --use_bu_loss \
         --buloss_option B \
         --learn_tau \
         --alpha 1.0 \
         --tau_init 1.0 \
         --boundary_radius 1 \
         --ckpt_dir ckpt/ckpt_Cataract1k \
         --ckpt GumbelTopK_BULoss \
         --description "Cataract1k_GumbelTopK and BULoss - ratio 0.5"
